    SUM(ss_coupon_amt) / COUNT(ss_coupon_amt) AS agg3,
    SUM(ss_sales_price) / COUNT(ss_sales_price) AS agg4
FROM (
    SELECT 
        ss_quantity, ss_list_price, ss_coupon_amt, ss_sales_price,
        ss_cdemo_sk, ss_sold_date_sk, ss_store_sk, ss_item_sk
    FROM store_sales
) AS store_sales
JOIN (
    SELECT cd_demo_sk
    FROM customer_demographics
    WHERE cd_gender = 'F' 
      AND cd_marital_status = 'W'
      AND cd_education_status = 'College'
) AS cd ON store_sales.ss_cdemo_sk = cd.cd_demo_sk
JOIN (
    SELECT d_date_sk
    FROM date_dim
    WHERE d_year = 1999
) AS dd ON store_sales.ss_sold_date_sk = dd.d_date_sk
JOIN (
    SELECT s_store_sk, s_state
    FROM store
    WHERE s_state = 'WI'
) AS s ON store_sales.ss_store_sk = s.s_store_sk
JOIN (
    SELECT i_item_sk, i_item_id
    FROM item
    WHERE i_category = 'Music'
) AS i ON store_sales.ss_item_sk = i.i_item_sk
GROUP BY ROLLUP (i_item_id, s_state)
ORDER BY i_item_id, s_state
LIMIT 100;
