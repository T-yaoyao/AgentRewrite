    FROM customer_total_return
    GROUP BY ctr_store_sk
)
SELECT c_customer_id
FROM customer_total_return ctr1
JOIN store_avg sa ON ctr1.ctr_store_sk = sa.ctr_store_sk AND ctr1.ctr_total_return > sa.threshold
JOIN store s ON s.s_store_sk = ctr1.ctr_store_sk
JOIN customer c ON ctr1.ctr_customer_sk = c.c_customer_sk
JOIN customer_demographics cd ON c.c_current_cdemo_sk = cd.cd_demo_sk
WHERE ctr1.ctr_reason_sk BETWEEN 28 AND 31
