JOIN (
    select ctr_store_sk, avg(ctr_total_return)*1.2 as avg_total_return 
    from customer_total_return 
    group by ctr_store_sk
) ctr2_avg ON ctr1.ctr_store_sk = ctr2_avg.ctr_store_sk AND ctr1.ctr_reason_sk BETWEEN 28 AND 31
JOIN store ON s_store_sk = ctr1.ctr_store_sk AND s_state IN ('MI', 'NC', 'WI')
JOIN customer ON ctr1.ctr_customer_sk = c_customer_sk 
    AND c_birth_month = 5 
    AND c_birth_year BETWEEN 1950 AND 1956
JOIN customer_demographics ON c_current_cdemo_sk = cd_demo_sk 
    AND cd_marital_status IN ('W', 'W') 
    AND cd_education_status IN ('4 yr Degree', 'College')

order by c_customer_id
