) AS date_dim

    SELECT ss_sold_date_sk, ss_item_sk, ss_customer_sk, ss_store_sk, ss_ext_sales_price
    FROM store_sales
    WHERE ss_wholesale_cost BETWEEN 57 AND 77
) AS store_sales ON date_dim.d_date_sk = store_sales.ss_sold_date_sk
JOIN (
    SELECT i_item_sk, i_brand_id, i_brand, i_manufact_id, i_manufact
    FROM item
    WHERE i_category = 'Books'
) AS item ON store_sales.ss_item_sk = item.i_item_sk
JOIN (
    SELECT c_customer_sk, c_current_addr_sk
    FROM customer
    WHERE c_birth_month = 10
) AS customer ON store_sales.ss_customer_sk = customer.c_customer_sk
JOIN (
    SELECT ca_address_sk, ca_zip
    FROM customer_address
    WHERE ca_state = 'IN'
) AS customer_address ON customer.c_current_addr_sk = customer_address.ca_address_sk
JOIN (
    SELECT s_store_sk, s_zip
    FROM store
) AS store ON store_sales.ss_store_sk = store.s_store_sk
WHERE SUBSTRING(customer_address.ca_zip, 1, 5) <> SUBSTRING(store.s_zip, 1, 5)
GROUP BY i_brand, i_brand_id, i_manufact_id, i_manufact
ORDER BY ext_price DESC, i_brand, i_brand_id, i_manufact_id, i_manufact
