),

),

  WHERE ca_state = 'IN'
),
filtered_store AS (

  FROM store
),
filtered_item AS (
  SELECT i_item_sk, i_brand_id, i_brand, i_manufact_id, i_manufact
  FROM item
  WHERE i_category = 'Books'
)
SELECT i_brand_id brand_id, i_brand brand, i_manufact_id, i_manufact,
  SUM(ss_ext_sales_price) ext_price
FROM filtered_date_dim
JOIN filtered_store_sales ON filtered_date_dim.d_date_sk = filtered_store_sales.ss_sold_date_sk
JOIN filtered_item ON filtered_store_sales.ss_item_sk = filtered_item.i_item_sk
JOIN filtered_customer ON filtered_store_sales.ss_customer_sk = filtered_customer.c_customer_sk
JOIN filtered_customer_address ON filtered_customer.c_current_addr_sk = filtered_customer_address.ca_address_sk
JOIN filtered_store ON filtered_store_sales.ss_store_sk = filtered_store.s_store_sk
WHERE substring(filtered_customer_address.ca_zip, 1, 5) <> substring(filtered_store.s_zip, 1, 5)
GROUP BY i_brand, i_brand_id, i_manufact_id, i_manufact
ORDER BY ext_price DESC, i_brand, i_brand_id, i_manufact_id, i_manufact
LIMIT 100;
