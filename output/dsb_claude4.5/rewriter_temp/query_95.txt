    cd.cd_gender,

    COUNT(*) AS cnt1,
    cd.cd_purchase_estimate,
    COUNT(*) AS cnt2,
    cd.cd_credit_rating,
    COUNT(*) AS cnt3
FROM customer c
JOIN customer_address ca ON c.c_current_addr_sk = ca.ca_address_sk AND ca.ca_state IN ('MS','NM','TN')
JOIN customer_demographics cd ON cd.cd_demo_sk = c.c_current_cdemo_sk
    AND cd.cd_marital_status IN ('W','S')
    AND cd.cd_education_status IN ('College','Advanced Degree')
LEFT JOIN store_sales_cte s ON c.c_customer_sk = s.ss_customer_sk
LEFT JOIN web_sales_cte w ON c.c_customer_sk = w.ws_bill_customer_sk
LEFT JOIN catalog_sales_cte cs ON c.c_customer_sk = cs.cs_ship_customer_sk
WHERE s.ss_customer_sk IS NOT NULL
    AND w.ws_bill_customer_sk IS NULL
    AND cs.cs_ship_customer_sk IS NULL
GROUP BY
