    w_warehouse_sk,
    i_item_sk,

      w_warehouse_sk,
      i_item_sk,
      d_moy,
      stddev_samp(inv_quantity_on_hand) AS stdev,
      AVG(inv_quantity_on_hand) AS mean
    FROM 
      inventory,
      item,
      warehouse,
      date_dim
    WHERE 
      inv_item_sk = i_item_sk
      AND inv_warehouse_sk = w_warehouse_sk
      AND inv_date_sk = d_date_sk
      AND d_year = 2001
      AND i_category IN ('Shoes', 'Sports')
      AND i_manager_id BETWEEN 81 AND 100
      AND inv_quantity_on_hand BETWEEN 30 AND 230
    GROUP BY 
      w_warehouse_name,
      w_warehouse_sk,
      i_item_sk,
      d_moy
  ) AS foo
  WHERE mean <> 0 AND stdev/mean > 1
)
