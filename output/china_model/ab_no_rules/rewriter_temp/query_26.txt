Rewritten: WITH distinct_jobs AS (SELECT DISTINCT JOB FROM EMP WHERE ENAME = 'A')
SELECT JOB FROM distinct_jobs
UNION
SELECT NULL::VARCHAR WHERE EXISTS (
    SELECT 1
    FROM DEPT d
    LEFT JOIN distinct_jobs j ON d.NAME = j.JOB
    WHERE j.JOB IS NULL
    LIMIT 1
)
