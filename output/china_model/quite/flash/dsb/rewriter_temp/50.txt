WITH customer_total_return AS (
    SELECT
        wr.wr_returning_customer_sk AS ctr_customer_sk,
        ra.ca_state AS ctr_state,
        wr.wr_reason_sk AS ctr_reason_sk,
        SUM(wr.wr_return_amt) AS ctr_total_return
    FROM web_returns wr
    JOIN date_dim d
        ON wr.wr_returned_date_sk = d.d_date_sk
    JOIN customer_address ra
        ON wr.wr_returning_addr_sk = ra.ca_address_sk
    JOIN item i

    WHERE d.d_year = 1998
      AND i.i_manager_id BETWEEN 40 AND 49
      AND wr.wr_return_amt / wr.wr_return_quantity BETWEEN 3 AND 32
    GROUP BY
        wr.wr_returning_customer_sk,
        ra.ca_state,
        wr.wr_reason_sk
),
state_avg AS (
    SELECT
        ctr_state,
        AVG(ctr_total_return) * 1.2 AS ctr_avg_threshold
