Original: select ps_partkey, sum(ps_supplycost * ps_availqty) as value from partsupp, supplier, nation where ps_suppkey = s_suppkey and s_nationkey = n_nationkey and n_name = 'ROMANIA' group by ps_partkey having sum(ps_supplycost * ps_availqty) > ( select sum(ps_supplycost * ps_availqty) * 0.0001000000 from partsupp, supplier, nation where ps_suppkey = s_suppkey and s_nationkey = n_nationkey and n_name = 'ROMANIA' ) order by value desc;
Rewritten: WITH partsupp_romania AS (
  SELECT ps_partkey, ps_supplycost * ps_availqty AS cost_value
  FROM partsupp, supplier, nation
  WHERE ps_suppkey = s_suppkey
    AND s_nationkey = n_nationkey
    AND n_name = 'ROMANIA'
),
part_totals AS (
  SELECT ps_partkey, SUM(cost_value) AS value
  FROM partsupp_romania
  GROUP BY ps_partkey
),
threshold AS (
  SELECT SUM(cost_value) * 0.0001000000 AS total_value
  FROM partsupp_romania
)
SELECT pt.ps_partkey, pt.value
FROM part_totals AS pt, threshold AS th
WHERE pt.value > th.total_value
ORDER BY pt.value DESC;
