  select cs.cs_item_sk, cs.cs_ext_discount_amt, cs.cs_list_price, cs.cs_sales_price

    and d.d_date_sk = cs.cs_sold_date_sk
),
avg_disc as (
  select c.cs_item_sk, avg(c.cs_ext_discount_amt) as avg_ext_discount
  from cs_dated c
  where c.cs_list_price between 260 and 289
    and c.cs_sales_price / c.cs_list_price between 0.75 and 0.95
  group by c.cs_item_sk
)
select sum(cs.cs_ext_discount_amt) as "excess discount amount"
from cs_dated cs
join item i on i.i_item_sk = cs.cs_item_sk
join avg_disc a on a.cs_item_sk = cs.cs_item_sk
where (i.i_manufact_id in (295, 363, 372, 688, 719) or i.i_manager_id between 11 and 40)
  and cs.cs_ext_discount_amt > 1.3 * a.avg_ext_discount
order by sum(cs.cs_ext_discount_amt)
limit 100
