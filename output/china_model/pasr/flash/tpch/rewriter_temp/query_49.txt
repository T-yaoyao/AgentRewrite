Rewritten: WITH big_order AS (
  SELECT l_orderkey
  FROM lineitem
  GROUP BY l_orderkey
  HAVING SUM(l_quantity) > 315
),
lineitem_agg AS (
  SELECT l_orderkey, SUM(l_quantity) AS sum_qty
  FROM lineitem
  GROUP BY l_orderkey
)
SELECT c.c_name, c.c_custkey, o.o_orderkey, o.o_orderdate, o.o_totalprice, a.sum_qty
FROM customer c
JOIN orders o ON c.c_custkey = o.o_custkey
JOIN lineitem_agg a ON o.o_orderkey = a.l_orderkey
JOIN big_order b ON o.o_orderkey = b.l_orderkey
ORDER BY o.o_totalprice DESC, o.o_orderdate
LIMIT 100;
