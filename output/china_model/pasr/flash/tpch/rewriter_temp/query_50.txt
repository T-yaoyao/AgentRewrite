SELECT c_name, c_custkey, o_orderkey, o_orderdate, o_totalprice, lineitem_agg.sum
FROM customer, orders, lineitem_agg, big_orders
WHERE c_custkey = o_custkey
  AND o_orderkey = lineitem_agg.l_orderkey
  AND o_orderkey = big_orders.l_orderkey
ORDER BY o_totalprice DESC, o_orderdate
LIMIT 100;
