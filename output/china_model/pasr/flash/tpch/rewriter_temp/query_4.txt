Rewritten: WITH p_filtered AS (
    SELECT p_partkey, p_mfgr
    FROM part
    WHERE p_size = 6 AND p_type LIKE '%NICKEL'
),
europe AS (
    SELECT n_nationkey, n_name
    FROM nation
    JOIN region ON n_regionkey = r_regionkey
    WHERE r_name = 'EUROPE'
),
min_cost AS (
    SELECT ps.ps_partkey, MIN(ps.ps_supplycost) AS min_supplycost
    FROM partsupp ps
    JOIN p_filtered ON p_filtered.p_partkey = ps.ps_partkey
    JOIN supplier s ON s.s_suppkey = ps.ps_suppkey
    JOIN europe ON s.s_nationkey = europe.n_nationkey
    GROUP BY ps.ps_partkey
)
SELECT s_acctbal, s_name, europe.n_name AS n_name, p_partkey, p_mfgr, s_address, s_phone, s_comment
FROM p_filtered
JOIN partsupp ON p_filtered.p_partkey = partsupp.ps_partkey
JOIN supplier ON supplier.s_suppkey = partsupp.ps_suppkey
JOIN europe ON supplier.s_nationkey = europe.n_nationkey
JOIN min_cost ON min_cost.ps_partkey = p_filtered.p_partkey
    AND partsupp.ps_supplycost = min_cost.min_supplycost
ORDER BY s_acctbal DESC, n_name, s_name, p_partkey
LIMIT 100;
