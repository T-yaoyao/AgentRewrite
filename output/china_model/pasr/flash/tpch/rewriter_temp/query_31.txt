Original: select ps_partkey, sum(ps_supplycost * ps_availqty) as value from partsupp, supplier, nation where ps_suppkey = s_suppkey and s_nationkey = n_nationkey and n_name = 'ROMANIA' group by ps_partkey having sum(ps_supplycost * ps_availqty) > ( select sum(ps_supplycost * ps_availqty) * 0.0001000000 from partsupp, supplier, nation where ps_suppkey = s_suppkey and s_nationkey = n_nationkey and n_name = 'ROMANIA' ) order by value desc;
Rewritten: WITH n_f AS (
  SELECT n_nationkey
  FROM nation
  WHERE n_name = 'ROMANIA'
),
ps_f AS (
  SELECT ps.ps_partkey, ps.ps_supplycost, ps.ps_availqty
  FROM partsupp AS ps
  JOIN supplier AS s ON ps.ps_suppkey = s.s_suppkey
  JOIN n_f ON s.s_nationkey = n_f.n_nationkey
)
SELECT ps_partkey, SUM(ps_supplycost * ps_availqty) AS value
FROM ps_f
GROUP BY ps_partkey
HAVING SUM(ps_supplycost * ps_availqty) > (
  SELECT SUM(ps_supplycost * ps_availqty) * 0.0001000000
  FROM ps_f
)
ORDER BY value DESC;
