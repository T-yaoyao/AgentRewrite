Original: select s_name, s_address from supplier, nation where s_suppkey in ( select ps_suppkey from partsupp where ps_partkey in ( select p_partkey from part where p_name like 'spring%' ) and ps_availqty > ( select 0.5 * sum(l_quantity) from lineitem where l_partkey = ps_partkey and l_suppkey = ps_suppkey and l_shipdate >= date '1997-01-01' and l_shipdate < date '1997-01-01' + interval '1' year ) ) and s_nationkey = n_nationkey and n_name = 'KENYA' order by s_name;
Rewritten: WITH lsum AS (
  SELECT l.l_partkey, l.l_suppkey, 0.5 * SUM(l.l_quantity) AS half_sum_qty
  FROM lineitem AS l
  WHERE l.l_shipdate >= DATE '1997-01-01' AND l.l_shipdate < DATE '1998-01-01'
    AND l.l_partkey IN (SELECT p_partkey FROM part WHERE p_name LIKE 'spring%')
  GROUP BY l.l_partkey, l.l_suppkey
),
ps_filtered AS (
  SELECT ps.ps_partkey, ps.ps_suppkey, ps.ps_availqty
  FROM partsupp AS ps
  WHERE ps.ps_partkey IN (SELECT p_partkey FROM part WHERE p_name LIKE 'spring%')
),
nation_filtered AS (
  SELECT n.n_nationkey
  FROM nation AS n
  WHERE n.n_name = 'KENYA'
)
SELECT s1.s_name, s1.s_address
FROM (
  SELECT s.s_suppkey, s.s_name, s.s_address, s.s_nationkey
  FROM supplier AS s
  WHERE s.s_suppkey IN (
    SELECT psf.ps_suppkey
    FROM ps_filtered AS psf
    JOIN lsum ON lsum.l_partkey = psf.ps_partkey AND lsum.l_suppkey = psf.ps_suppkey
    WHERE psf.ps_availqty > lsum.half_sum_qty
  )
) AS s1
JOIN nation_filtered AS nf ON s1.s_nationkey = nf.n_nationkey
ORDER BY s1.s_name;
