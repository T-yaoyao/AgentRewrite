Original: select ps_partkey, sum(ps_supplycost * ps_availqty) as value from partsupp, supplier, nation where ps_suppkey = s_suppkey and s_nationkey = n_nationkey and n_name = 'IRAN' group by ps_partkey having sum(ps_supplycost * ps_availqty) > ( select sum(ps_supplycost * ps_availqty) * 0.0001000000 from partsupp, supplier, nation where ps_suppkey = s_suppkey and s_nationkey = n_nationkey and n_name = 'IRAN' ) order by value desc;
Rewritten: WITH iran_supply AS (
  SELECT ps_partkey, ps_supplycost, ps_availqty
  FROM partsupp
  JOIN supplier s ON ps_suppkey = s.s_suppkey
  JOIN (SELECT n_nationkey FROM nation WHERE n_name = 'IRAN') AS n_filtered
    ON s_nationkey = n_filtered.n_nationkey
)
SELECT ps_partkey, SUM(ps_supplycost * ps_availqty) AS value
FROM iran_supply
GROUP BY ps_partkey
HAVING SUM(ps_supplycost * ps_availqty) > (
  SELECT SUM(ps_supplycost * ps_availqty) * 0.0001000000
  FROM iran_supply
)
ORDER BY value DESC;
