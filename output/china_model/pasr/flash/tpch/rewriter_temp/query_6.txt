Original: select s_acctbal, s_name, n_name, p_partkey, p_mfgr, s_address, s_phone, s_comment from part, supplier, partsupp, nation, region where p_partkey = ps_partkey and s_suppkey = ps_suppkey and p_size = 3 and p_type like '%NICKEL' and s_nationkey = n_nationkey and n_regionkey = r_regionkey and r_name = 'AMERICA' and ps_supplycost = ( select min(ps_supplycost) from partsupp, supplier, nation, region where p_partkey = ps_partkey and s_suppkey = ps_suppkey and s_nationkey = n_nationkey and n_regionkey = r_regionkey and r_name = 'AMERICA' ) order by s_acctbal desc, n_name, s_name, p_partkey limit 100;
Rewritten: WITH filtered_part AS (
    SELECT p_partkey, p_mfgr
    FROM part
    WHERE p_size = 3 AND p_type LIKE '%NICKEL'
),
america_supplier AS (
    SELECT s.s_suppkey, s.s_name, s.s_address, s.s_phone, s.s_acctbal, s.s_comment, n.n_name
    FROM supplier s
    JOIN nation n ON s.s_nationkey = n.n_nationkey
    JOIN region r ON n.n_regionkey = r.r_regionkey
    WHERE r.r_name = 'AMERICA'
),
min_cost AS (
    SELECT ps.ps_partkey, MIN(ps.ps_supplycost) AS min_supplycost
    FROM partsupp ps
    JOIN america_supplier asup ON ps.ps_suppkey = asup.s_suppkey
    JOIN filtered_part fp ON ps.ps_partkey = fp.p_partkey
    GROUP BY ps.ps_partkey
)
SELECT asup.s_acctbal, asup.s_name, asup.n_name, fp.p_partkey, fp.p_mfgr, asup.s_address, asup.s_phone, asup.s_comment
FROM filtered_part fp
JOIN partsupp ps ON fp.p_partkey = ps.ps_partkey
JOIN america_supplier asup ON ps.ps_suppkey = asup.s_suppkey
JOIN min_cost mc ON mc.ps_partkey = fp.p_partkey AND ps.ps_supplycost = mc.min_supplycost
ORDER BY s_acctbal DESC, n_name, s_name, p_partkey
LIMIT 100;
