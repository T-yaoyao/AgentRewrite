  ON p.p_partkey = ps.ps_partkey
JOIN (SELECT s_suppkey, s_acctbal, s_name, s_address, s_phone, s_comment, s_nationkey FROM supplier) AS s
  ON s.s_suppkey = ps.ps_suppkey

JOIN min_cost ON p.p_partkey = min_cost.ps_partkey AND ps.ps_supplycost = min_cost.min_supplycost
