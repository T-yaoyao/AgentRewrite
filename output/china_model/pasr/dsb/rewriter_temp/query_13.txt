    SELECT d_date_sk

SELECT avg(ss.ss_quantity),
       avg(ss.ss_ext_sales_price),
       avg(ss.ss_ext_wholesale_cost),
       sum(ss.ss_ext_wholesale_cost)
FROM store_sales ss
JOIN store st ON st.s_store_sk = ss.ss_store_sk
JOIN date_filtered dd ON ss.ss_sold_date_sk = dd.d_date_sk
JOIN (
    SELECT hd_demo_sk, hd_dep_count
    FROM household_demographics
    WHERE hd_dep_count IN (1, 3)
) hd ON ss.ss_hdemo_sk = hd.hd_demo_sk
JOIN (
    SELECT cd_demo_sk, cd_marital_status, cd_education_status
    FROM customer_demographics
    WHERE cd_marital_status IN ('U', 'W', 'S')
      AND cd_education_status IN ('College', '2 yr Degree')
) cd ON ss.ss_cdemo_sk = cd.cd_demo_sk
JOIN (
    SELECT ca_address_sk, ca_state
    FROM customer_address
    WHERE ca_country = 'United States'
      AND ca_state IN ('IN', 'NM', 'VA', 'MT', 'OH', 'OR', 'GA', 'IL', 'TX')
) ca ON ss.ss_addr_sk = ca.ca_address_sk
WHERE (
        (cd.cd_marital_status = 'U'
         AND cd.cd_education_status = 'College'
         AND ss.ss_sales_price BETWEEN 100.00 AND 150.00
         AND hd.hd_dep_count = 3)
     OR (cd.cd_marital_status = 'W'
         AND cd.cd_education_status = '2 yr Degree'
         AND ss.ss_sales_price BETWEEN 50.00 AND 100.00
         AND hd.hd_dep_count = 1)
     OR (cd.cd_marital_status = 'S'
         AND cd.cd_education_status = 'College'
         AND ss.ss_sales_price BETWEEN 150.00 AND 200.00
         AND hd.hd_dep_count = 1)
      )
  AND (
        (ca.ca_state IN ('IN', 'NM', 'VA')
         AND ss.ss_net_profit BETWEEN 100 AND 200)
     OR (ca.ca_state IN ('MT', 'OH', 'OR')
         AND ss.ss_net_profit BETWEEN 150 AND 300)
     OR (ca.ca_state IN ('GA', 'IL', 'TX')
         AND ss.ss_net_profit BETWEEN 50 AND 250)
      );
