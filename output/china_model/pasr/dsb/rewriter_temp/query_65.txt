           agg.i_item_sk, agg.d_moy, agg.stdev, agg.mean

             stddev_samp(inv_quantity_on_hand) stdev,
             sum(inv_quantity_on_hand)::numeric / nullif(count(inv_quantity_on_hand), 0) mean
      from inventory, date_dim
      where inv_date_sk = d_date_sk
        and d_year = 2000
        and d_moy in (1, 2)
        and inv_quantity_on_hand between 376 and 576
      group by inv_item_sk, inv_warehouse_sk, d_moy
    ) agg, item, warehouse
    where agg.i_item_sk = item.i_item_sk
      and agg.w_warehouse_sk = warehouse.w_warehouse_sk
      and i_category IN ('Books', 'Men')
      and i_manager_id BETWEEN 81 and 100
  ) foo
  where case mean when 0 then 0 else stdev/mean end > 1
)
