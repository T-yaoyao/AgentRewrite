  SELECT iss.i_brand_id AS brand_id, iss.i_class_id AS class_id, iss.i_category_id AS category_id
  FROM store_sales
  JOIN item iss ON ss_item_sk = iss.i_item_sk
  JOIN date_dim d1 ON ss_sold_date_sk = d1.d_date_sk
  WHERE d1.d_year BETWEEN 1998 AND 2000
    AND iss.i_category IN ('Books', 'Home', 'Sports')
    AND iss.i_manager_id BETWEEN 14 AND 23
    AND ss_wholesale_cost BETWEEN 78 AND 98
),
cs_brand_class AS (

  FROM catalog_sales
  JOIN item ics ON cs_item_sk = ics.i_item_sk
  JOIN date_dim d2 ON cs_sold_date_sk = d2.d_date_sk
  WHERE d2.d_year BETWEEN 1998 AND 2000
    AND ics.i_category IN ('Books', 'Home', 'Sports')
    AND ics.i_manager_id BETWEEN 14 AND 23
    AND cs_wholesale_cost BETWEEN 78 AND 98

  JOIN date_dim d3 ON ws_sold_date_sk = d3.d_date_sk

),

  WHERE i_category IN ('Books', 'Home', 'Sports')

),

  WHERE d_year BETWEEN 1998 AND 2000
    AND ss_wholesale_cost BETWEEN 78 AND 98
),
cs_qty_price AS (
  SELECT cs_quantity AS quantity, cs_list_price AS list_price
  FROM catalog_sales
  JOIN date_dim ON cs_sold_date_sk = d_date_sk
  WHERE d_year BETWEEN 1998 AND 2000
    AND cs_wholesale_cost BETWEEN 78 AND 98
),
ws_qty_price AS (
  SELECT ws_quantity AS quantity, ws_list_price AS list_price
  FROM web_sales
  JOIN date_dim ON ws_sold_date_sk = d_date_sk
  WHERE ws_wholesale_cost BETWEEN 78 AND 98
    AND d_year BETWEEN 1998 AND 2000
),
avg_sales AS (
  SELECT sum(total_sales) / NULLIF(sum(sales_cnt), 0) AS average_sales
  FROM (
    SELECT sum(quantity * list_price) AS total_sales, count(quantity * list_price) AS sales_cnt FROM ss_qty_price
    UNION ALL
    SELECT sum(quantity * list_price), count(quantity * list_price) FROM cs_qty_price
    UNION ALL
    SELECT sum(quantity * list_price), count(quantity * list_price) FROM ws_qty_price
  ) b
),
ty_week_seq AS (
  SELECT d_week_seq FROM date_dim WHERE d_year = 1999 AND d_moy = 12 AND d_dom = 13
),
ly_week_seq AS (
  SELECT d_week_seq FROM date_dim WHERE d_year = 1998 AND d_moy = 12 AND d_dom = 13
),
ty_item_sales AS (
  SELECT ss_item_sk, sum(ss_quantity * ss_list_price) AS sales, count(*) AS number_sales
  FROM store_sales
  JOIN date_dim dd ON ss_sold_date_sk = dd.d_date_sk
  JOIN ty_week_seq tw ON dd.d_week_seq = tw.d_week_seq
  WHERE ss_wholesale_cost BETWEEN 78 AND 98
  GROUP BY ss_item_sk
),
ly_item_sales AS (
  SELECT ss_item_sk, sum(ss_quantity * ss_list_price) AS sales, count(*) AS number_sales
  FROM store_sales
  JOIN date_dim dd ON ss_sold_date_sk = dd.d_date_sk
  JOIN ly_week_seq lw ON dd.d_week_seq = lw.d_week_seq
  WHERE ss_wholesale_cost BETWEEN 78 AND 98
  GROUP BY ss_item_sk
),
this_year AS (
  SELECT 'store' AS channel, i_brand_id, i_class_id, i_category_id,
         sum(tys.sales) AS sales, sum(tys.number_sales) AS number_sales
  FROM ty_item_sales tys
  JOIN item ON tys.ss_item_sk = i_item_sk
  JOIN cross_items ci ON tys.ss_item_sk = ci.ss_item_sk
  WHERE i_category IN ('Books', 'Home', 'Sports')
    AND i_manager_id BETWEEN 14 AND 23
  GROUP BY i_brand_id, i_class_id, i_category_id
  HAVING sum(tys.sales) > (SELECT average_sales FROM avg_sales)

last_year AS (
  SELECT 'store' AS channel, i_brand_id, i_class_id, i_category_id,

  JOIN cross_items ci ON lys.ss_item_sk = ci.ss_item_sk
  WHERE i_category IN ('Books', 'Home', 'Sports')
    AND i_manager_id BETWEEN 14 AND 23
  GROUP BY i_brand_id, i_class_id, i_category_id
  HAVING sum(lys.sales) > (SELECT average_sales FROM avg_sales)
)
