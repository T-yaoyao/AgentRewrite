),
filtered_partsupp AS (
    SELECT ps_partkey, ps_supplycost, ps_availqty
    FROM partsupp
    WHERE ps_suppkey IN (SELECT s_suppkey FROM jordan_suppliers)
),
aggregated AS (
    SELECT ps_partkey, SUM(ps_supplycost * ps_availqty) AS total_value
    FROM filtered_partsupp
    GROUP BY ps_partkey
),
threshold AS (
    SELECT SUM(ps_supplycost * ps_availqty) * 0.0001000000 AS threshold_value
    FROM filtered_partsupp
)
SELECT a.ps_partkey, a.total_value AS value
FROM aggregated a
CROSS JOIN threshold t
WHERE a.total_value > t.threshold_value
ORDER BY value DESC;
