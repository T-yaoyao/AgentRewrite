total_value AS (
    SELECT sum(ps_supplycost * ps_availqty) AS total

)
SELECT ps_partkey, sum(ps_supplycost * ps_availqty) AS value
FROM iran_partsupp
GROUP BY ps_partkey
HAVING sum(ps_supplycost * ps_availqty) > (SELECT total * 0.0001000000 FROM total_value)
ORDER BY value DESC;
