Rewritten: SELECT s_name, s_address
FROM supplier
JOIN (
    SELECT DISTINCT partsupp.ps_suppkey
    FROM partsupp
    JOIN part ON partsupp.ps_partkey = part.p_partkey AND part.p_name LIKE 'burlywood%'
    JOIN (
        SELECT l_partkey, l_suppkey, 0.5 * SUM(l_quantity) AS half_sum
        FROM lineitem
        WHERE l_shipdate >= DATE '1994-01-01'
          AND l_shipdate < DATE '1994-01-01' + INTERVAL '1' YEAR
        GROUP BY l_partkey, l_suppkey
    ) AS agg ON partsupp.ps_partkey = agg.l_partkey AND partsupp.ps_suppkey = agg.l_suppkey
    WHERE partsupp.ps_availqty > agg.half_sum
) AS sub ON supplier.s_suppkey = sub.ps_suppkey
JOIN nation ON supplier.s_nationkey = nation.n_nationkey AND nation.n_name = 'ETHIOPIA'
ORDER BY s_name;
