  SELECT n_name AS nation,

  FROM (SELECT p_partkey FROM part WHERE p_name LIKE '%medium%') AS part
  JOIN (SELECT l_suppkey, l_partkey, l_orderkey, l_extendedprice, l_discount, l_quantity FROM lineitem) AS lineitem ON part.p_partkey = lineitem.l_partkey

  JOIN (SELECT s_suppkey, s_nationkey FROM supplier) AS supplier ON supplier.s_suppkey = lineitem.l_suppkey
  JOIN (SELECT n_nationkey, n_name FROM nation) AS nation ON supplier.s_nationkey = nation.n_nationkey
) AS profit
GROUP BY nation, o_year
ORDER BY nation, o_year DESC;
