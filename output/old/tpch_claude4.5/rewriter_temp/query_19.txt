  JOIN (
    SELECT o_orderkey, o_custkey
    FROM orders
  ) AS o ON l.l_orderkey = o.o_orderkey
  JOIN (
    SELECT c_custkey, c_nationkey
    FROM customer
    WHERE EXISTS (
      SELECT 1 FROM nation WHERE n_nationkey = c_nationkey AND n_name IN ('CANADA', 'CHINA')
    )
  ) AS c ON o.o_custkey = c.c_custkey
  JOIN (
    SELECT n_nationkey, n_name
    FROM nation
    WHERE n_name IN ('CANADA', 'CHINA')

    SELECT n_nationkey, n_name
    FROM nation

  ) AS n2 ON c.c_nationkey = n2.n_nationkey

     OR (n1.n_name = 'CHINA' AND n2.n_name = 'CANADA')
