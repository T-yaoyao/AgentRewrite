Rewritten: WITH date_range AS (
  SELECT d_date_sk
  FROM date_dim
  WHERE d_date >= '2001-01-31'
    AND d_date <= DATE '2001-01-31' + INTERVAL '90' day
),
filtered_items AS (
  SELECT i_item_sk
  FROM item
  WHERE (i_manufact_id BETWEEN 749 AND 948)
     OR (i_category IN ('Books', 'Home', 'Sports'))
),
avg_discount AS (
  SELECT ws.ws_item_sk,
         1.3 * AVG(ws.ws_ext_discount_amt) AS threshold
  FROM web_sales ws
  INNER JOIN date_range dr ON ws.ws_sold_date_sk = dr.d_date_sk
  INNER JOIN filtered_items fi ON ws.ws_item_sk = fi.i_item_sk
  WHERE ws.ws_wholesale_cost BETWEEN 57 AND 77
    AND ws.ws_sales_price / NULLIF(ws.ws_list_price, 0) BETWEEN 0.17 AND 0.32
  GROUP BY ws.ws_item_sk
)
SELECT SUM(ws.ws_ext_discount_amt) AS "Excess Discount Amount"
FROM web_sales ws
INNER JOIN date_range dr ON ws.ws_sold_date_sk = dr.d_date_sk
INNER JOIN filtered_items fi ON ws.ws_item_sk = fi.i_item_sk
INNER JOIN avg_discount ad ON ws.ws_item_sk = ad.ws_item_sk
WHERE ws.ws_wholesale_cost BETWEEN 57 AND 77
  AND ws.ws_ext_discount_amt > ad.threshold
ORDER BY SUM(ws.ws_ext_discount_amt)
LIMIT 100;
