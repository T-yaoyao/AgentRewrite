Original: select s_name, s_address from supplier, nation where s_suppkey in ( select ps_suppkey from partsupp where ps_partkey in ( select p_partkey from part where p_name like 'burlywood%' ) and ps_availqty > ( select 0.5 * sum(l_quantity) from lineitem where l_partkey = ps_partkey and l_suppkey = ps_suppkey and l_shipdate >= date '1994-01-01' and l_shipdate < date '1994-01-01' + interval '1' year ) ) and s_nationkey = n_nationkey and n_name = 'ETHIOPIA' order by s_name;
Rewritten: SELECT s_name, s_address
FROM nation
INNER JOIN supplier ON s_nationkey = n_nationkey
WHERE n_name = 'ETHIOPIA'
  AND EXISTS (
    SELECT 1
    FROM part
    INNER JOIN partsupp ON ps_partkey = p_partkey
    LEFT JOIN (
      SELECT l_partkey, l_suppkey, 0.5 * SUM(l_quantity) AS threshold_qty
      FROM lineitem
      WHERE l_shipdate >= DATE '1994-01-01'
        AND l_shipdate < DATE '1994-01-01' + INTERVAL '1' YEAR
      GROUP BY l_partkey, l_suppkey
    ) agg_lineitem ON agg_lineitem.l_partkey = ps_partkey
                   AND agg_lineitem.l_suppkey = ps_suppkey
    WHERE p_name LIKE 'burlywood%'
      AND ps_suppkey = supplier.s_suppkey
      AND (agg_lineitem.threshold_qty IS NULL OR ps_availqty > agg_lineitem.threshold_qty)
  )
ORDER BY s_name;
