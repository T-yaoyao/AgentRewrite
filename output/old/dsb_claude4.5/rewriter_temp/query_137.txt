Rewritten: WITH common_ws AS (
    SELECT ws.ws_item_sk, ws.ws_ext_discount_amt, ws.ws_sales_price, ws.ws_list_price
    FROM web_sales ws
    JOIN date_dim d ON ws.ws_sold_date_sk = d.d_date_sk
    WHERE d.d_date BETWEEN '2001-01-31'::date AND '2001-05-01'::date
      AND ws.ws_wholesale_cost BETWEEN 57 AND 77
),
item_avg AS (
    SELECT ws_item_sk, 1.3 * AVG(ws_ext_discount_amt) AS avg_discount
    FROM common_ws
    WHERE ws_sales_price / ws_list_price BETWEEN 0.17 AND 0.32
    GROUP BY ws_item_sk
)
SELECT SUM(cws.ws_ext_discount_amt) AS "Excess Discount Amount"
FROM common_ws cws
JOIN item i ON cws.ws_item_sk = i.i_item_sk
  AND (i.i_manufact_id BETWEEN 749 AND 948 OR i.i_category IN ('Books', 'Home', 'Sports'))
JOIN item_avg ia ON cws.ws_item_sk = ia.ws_item_sk
WHERE cws.ws_ext_discount_amt > ia.avg_discount
ORDER BY SUM(cws.ws_ext_discount_amt)
LIMIT 100;
