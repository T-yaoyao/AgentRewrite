    where date_dim.d_year = 2001

      and item.i_category = 'Sports'
      and item.i_class = 'camping'
      and web_sales.ws_wholesale_cost between 18 and 48

),
my_revenue as (

  join store on customer_address.ca_county = store.s_county and customer_address.ca_state = store.s_state
