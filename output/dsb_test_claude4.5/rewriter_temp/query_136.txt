```sql
WITH date_filtered AS (
    SELECT d_date_sk
    FROM date_dim
    WHERE d_date BETWEEN '2002-02-11' AND CAST('2002-02-11' AS DATE) + INTERVAL '90' DAY
)
SELECT 
    SUM(ws_ext_discount_amt) AS "Excess Discount Amount"
FROM (
    SELECT ws_ext_discount_amt, ws_item_sk, ws_sold_date_sk
    FROM web_sales
    WHERE ws_wholesale_cost BETWEEN 68 AND 88
      AND ws_item_sk IS NOT NULL
      AND ws_sold_date_sk IS NOT NULL
) AS ws
JOIN (
    SELECT i_item_sk
    FROM item
    WHERE (i_manufact_id BETWEEN 394 AND 593 OR i_category IN ('Books', 'Home', 'Sports'))
      AND i_item_sk IS NOT NULL
) AS it ON ws.ws_item_sk = it.i_item_sk
JOIN date_filtered df ON ws.ws_sold_date_sk = df.d_date_sk
JOIN (
    SELECT 
        ws_item_sk, 
        1.3 * AVG(ws_ext_discount_amt) AS avg_discount
    FROM (
        SELECT ws_item_sk, ws_ext_discount_amt, ws_sold_date_sk
        FROM web_sales
        WHERE ws_wholesale_cost BETWEEN 68 AND 88
          AND ws_sales_price / ws_list_price BETWEEN 85 * 0.01 AND 100 * 0.01
          AND ws_item_sk IS NOT NULL
          AND ws_sold_date_sk IS NOT NULL
    ) AS ws_sub
    JOIN date_filtered df_sub ON ws_sub.ws_sold_date_sk = df_sub.d_date_sk
    GROUP BY ws_item_sk
) AS sub ON it.i_item_sk = sub.ws_item_sk
WHERE ws.ws_ext_discount_amt > sub.avg_discount
ORDER BY SUM(ws_ext_discount_amt)
LIMIT 100;
```
