Rewritten: WITH avg_balance AS (
  SELECT avg(c_acctbal) AS avg_acct
  FROM customer
  WHERE c_acctbal > 0.00

),
cust_filtered AS (
  SELECT 
    c_custkey,
    substring(c_phone from 1 for 2) AS cntrycode,
    c_acctbal
  FROM customer
  CROSS JOIN avg_balance
  WHERE substring(c_phone from 1 for 2) IN ('24', '31', '26', '34', '30', '22', '14')
    AND c_acctbal > avg_balance.avg_acct
)
SELECT 
  cf.cntrycode,
  count(*) AS numcust,
  sum(cf.c_acctbal) AS totacctbal
FROM cust_filtered cf
LEFT JOIN orders o ON o.o_custkey = cf.c_custkey
WHERE o.o_custkey IS NULL
GROUP BY cf.cntrycode
ORDER BY cf.cntrycode;
