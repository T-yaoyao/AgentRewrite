Rewritten: WITH avg_acctbal AS (
  SELECT avg(c_acctbal) AS avg_bal
  FROM customer
  WHERE c_acctbal > 0.00
    AND substring(c_phone from 1 for 2) IN ('15', '18', '31', '33', '32', '28', '16')
),
filtered_customers AS (
  SELECT substring(c_phone from 1 for 2) AS cntrycode,
         c_acctbal,
         c_custkey
  FROM customer, avg_acctbal
  WHERE substring(c_phone from 1 for 2) IN ('15', '18', '31', '33', '32', '28', '16')
    AND c_acctbal > avg_acctbal.avg_bal
)
SELECT cntrycode,
       count(*) AS numcust,
       sum(c_acctbal) AS totacctbal
FROM filtered_customers fc
WHERE NOT EXISTS (
  SELECT 1
  FROM orders
  WHERE o_custkey = fc.c_custkey
)
GROUP BY cntrycode
ORDER BY cntrycode;
