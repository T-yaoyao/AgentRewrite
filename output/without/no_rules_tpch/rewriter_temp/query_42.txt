Original: select 100.00 * sum(case when p_type like 'PROMO%' then l_extendedprice * (1 - l_discount) else 0 end) / sum(l_extendedprice * (1 - l_discount)) as promo_revenue from lineitem, part where l_partkey = p_partkey and l_shipdate >= date '1997-03-01' and l_shipdate < date '1997-03-01' + interval '1' month;
Rewritten: WITH filtered_lineitem AS (
  SELECT l_partkey, l_extendedprice, l_discount
  FROM lineitem
  WHERE l_shipdate >= DATE '1997-03-01'
    AND l_shipdate < DATE '1997-03-01' + INTERVAL '1' month
),
joined_data AS (
  SELECT 
    p.p_type,
    fl.l_extendedprice * (1 - fl.l_discount) AS revenue
  FROM filtered_lineitem fl
  INNER JOIN part p ON fl.l_partkey = p.p_partkey
),
aggregated_revenue AS (
  SELECT 
    SUM(CASE WHEN p_type LIKE 'PROMO%' THEN revenue ELSE 0 END) AS promo_revenue,
    SUM(revenue) AS total_revenue
  FROM joined_data
)
SELECT 100.00 * promo_revenue / total_revenue AS promo_revenue
FROM aggregated_revenue;
