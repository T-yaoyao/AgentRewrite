Rewritten: WITH romania_suppliers AS (

  WHERE n.n_name = 'ROMANIA'
),
partsupp_agg AS (
  SELECT ps.ps_partkey, SUM(ps.ps_supplycost * ps.ps_availqty) AS part_value
  FROM partsupp ps
  INNER JOIN romania_suppliers rs ON ps.ps_suppkey = rs.s_suppkey
  GROUP BY ps.ps_partkey
),
total_value AS (
  SELECT SUM(ps.ps_supplycost * ps.ps_availqty) * 0.0001000000 AS threshold
  FROM partsupp ps
  INNER JOIN romania_suppliers rs ON ps.ps_suppkey = rs.s_suppkey
)
SELECT pa.ps_partkey, pa.part_value AS value
FROM partsupp_agg pa
CROSS JOIN total_value tv
WHERE pa.part_value > tv.threshold
ORDER BY value DESC;
