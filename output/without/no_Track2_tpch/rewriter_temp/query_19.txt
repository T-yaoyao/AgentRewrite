Rewritten: WITH filtered_nation_pairs AS (

  FROM nation n1
  INNER JOIN nation n2 ON (n1.n_name = 'CANADA' AND n2.n_name = 'CHINA')
                       OR (n1.n_name = 'CHINA' AND n2.n_name = 'CANADA')
  WHERE n1.n_name IN ('CANADA', 'CHINA')
    AND n2.n_name IN ('CANADA', 'CHINA')
),
filtered_lineitem AS (
  SELECT l_suppkey, l_orderkey, l_extendedprice, l_discount,
         EXTRACT(YEAR FROM l_shipdate) AS l_year
  FROM lineitem
  WHERE l_shipdate BETWEEN DATE '1995-01-01' AND DATE '1996-12-31'
    AND l_suppkey IS NOT NULL
    AND l_orderkey IS NOT NULL
)
SELECT fnp.supp_nation, fnp.cust_nation, fl.l_year,
       SUM(fl.l_extendedprice * (1 - fl.l_discount)) AS revenue
FROM filtered_nation_pairs fnp
INNER JOIN supplier s ON s.s_nationkey = fnp.supp_nationkey
                      AND s.s_suppkey IS NOT NULL
INNER JOIN filtered_lineitem fl ON fl.l_suppkey = s.s_suppkey
INNER JOIN orders o ON o.o_orderkey = fl.l_orderkey
                    AND o.o_custkey IS NOT NULL
INNER JOIN customer c ON c.c_custkey = o.o_custkey
                      AND c.c_nationkey = fnp.cust_nationkey
GROUP BY fnp.supp_nation, fnp.cust_nation, fl.l_year
ORDER BY fnp.supp_nation, fnp.cust_nation, fl.l_year
