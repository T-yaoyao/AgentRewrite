        FROM region
        WHERE r_name = 'AMERICA'
    )
) AS nation_filtered
INNER JOIN supplier ON supplier.s_nationkey = nation_filtered.n_nationkey
INNER JOIN lineitem ON lineitem.l_suppkey = supplier.s_suppkey
INNER JOIN (
    SELECT o_orderkey, o_custkey
    FROM orders
    WHERE o_orderdate >= DATE '1995-01-01'
      AND o_orderdate < DATE '1995-01-01' + INTERVAL '1' YEAR
) AS orders_filtered ON orders_filtered.o_orderkey = lineitem.l_orderkey
INNER JOIN customer ON customer.c_custkey = orders_filtered.o_custkey
    AND customer.c_nationkey = nation_filtered.n_nationkey
GROUP BY n_name
ORDER BY revenue DESC;
