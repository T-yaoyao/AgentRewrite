Rewritten: WITH avg_acctbal AS (
    SELECT avg(c_acctbal) AS avg_bal
    FROM customer
    WHERE c_acctbal > 0.00
      AND substring(c_phone FROM 1 FOR 2) IN ('24', '31', '26', '34', '30', '22', '14')
),
filtered_customer AS (
    SELECT c_custkey,
           substring(c_phone FROM 1 FOR 2) AS cntrycode,
           c_acctbal
    FROM customer
    CROSS JOIN avg_acctbal
    WHERE substring(c_phone FROM 1 FOR 2) IN ('24', '31', '26', '34', '30', '22', '14')
      AND c_acctbal > avg_acctbal.avg_bal
),
customers_with_orders AS (
    SELECT DISTINCT o_custkey
    FROM orders
    WHERE o_custkey IS NOT NULL
)
SELECT fc.cntrycode,
       count(*) AS numcust,
       sum(fc.c_acctbal) AS totacctbal
FROM filtered_customer fc
LEFT JOIN customers_with_orders cwo ON fc.c_custkey = cwo.o_custkey
WHERE cwo.o_custkey IS NULL
GROUP BY fc.cntrycode
ORDER BY fc.cntrycode
