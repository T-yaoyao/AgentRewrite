Rewritten: WITH iran_suppliers AS (
  SELECT s.s_suppkey
  FROM supplier s
  JOIN nation n ON s.s_nationkey = n.n_nationkey
  WHERE n.n_name = 'IRAN'
),
partsupp_agg AS (
  SELECT 
    ps.ps_partkey,
    SUM(ps.ps_supplycost * ps.ps_availqty) AS part_value
  FROM partsupp ps
  JOIN iran_suppliers is_sup ON ps.ps_suppkey = is_sup.s_suppkey
  GROUP BY ps.ps_partkey
),
total_value AS (
  SELECT SUM(ps.ps_supplycost * ps.ps_availqty) * 0.0001000000 AS threshold
  FROM partsupp ps
  JOIN iran_suppliers is_sup ON ps.ps_suppkey = is_sup.s_suppkey
)
SELECT 
  pa.ps_partkey,
  pa.part_value AS value
FROM partsupp_agg pa
CROSS JOIN total_value tv
WHERE pa.part_value > tv.threshold
ORDER BY value DESC;
