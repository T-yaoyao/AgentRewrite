lineitem_agg AS (
  SELECT l_orderkey, SUM(l_quantity) AS sum_quantity

SELECT c.c_name, c.c_custkey, o.o_orderkey, o.o_orderdate, o.o_totalprice, la.sum_quantity
FROM customer c
JOIN orders o ON c.c_custkey = o.o_custkey
