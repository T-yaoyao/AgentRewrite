Rewritten: SELECT s.s_name, s.s_address
FROM (
    SELECT s_suppkey, s_name, s_address
    FROM supplier
    WHERE s_nationkey IN (SELECT n_nationkey FROM nation WHERE n_name = 'GERMANY')
) AS s
JOIN (
    SELECT DISTINCT ps_suppkey
    FROM (
        SELECT ps_suppkey, ps_partkey, ps_availqty
        FROM partsupp
        WHERE ps_partkey IN (SELECT p_partkey FROM part WHERE p_name LIKE 'burnished%')
    ) AS ps
    LEFT JOIN (
        SELECT l_partkey, l_suppkey, SUM(l_quantity) * 0.5 AS threshold_qty
        FROM lineitem
        WHERE l_shipdate >= DATE '1996-01-01'
          AND l_shipdate < DATE '1996-01-01' + INTERVAL '1' year
        GROUP BY l_partkey, l_suppkey
    ) AS l_agg
    ON ps.ps_partkey = l_agg.l_partkey AND ps.ps_suppkey = l_agg.l_suppkey
    WHERE ps.ps_availqty > COALESCE(l_agg.threshold_qty, 0)
) AS qualified_ps
ON s.s_suppkey = qualified_ps.ps_suppkey
ORDER BY s.s_name
