Rewritten: WITH jordan_suppliers AS (
  SELECT s.s_suppkey
  FROM supplier s
  JOIN nation n ON s.s_nationkey = n.n_nationkey
  WHERE n.n_name = 'JORDAN'
),
partsupp_agg AS (
  SELECT ps.ps_partkey, SUM(ps.ps_supplycost * ps.ps_availqty) AS value
  FROM partsupp ps
  JOIN jordan_suppliers js ON ps.ps_suppkey = js.s_suppkey
  GROUP BY ps.ps_partkey
),
threshold AS (
  SELECT SUM(ps.ps_supplycost * ps.ps_availqty) * 0.0001000000 AS threshold_value
  FROM partsupp ps
  JOIN jordan_suppliers js ON ps.ps_suppkey = js.s_suppkey
)
SELECT pa.ps_partkey, pa.value
FROM partsupp_agg pa
CROSS JOIN threshold t
WHERE pa.value > t.threshold_value
ORDER BY pa.value DESC;
