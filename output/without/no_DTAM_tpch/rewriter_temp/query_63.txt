        c_acctbal
    FROM customer
    CROSS JOIN avg_balance
    WHERE substring(c_phone FROM 1 FOR 2) IN ('15', '18', '31', '33', '32', '28', '16')
      AND c_acctbal > avg_balance.avg_bal
)
SELECT 
    fc.cntrycode,
    count(*) AS numcust,
    sum(fc.c_acctbal) AS totacctbal
FROM filtered_customers fc
LEFT JOIN orders o ON o.o_custkey = fc.c_custkey
WHERE o.o_custkey IS NULL
GROUP BY fc.cntrycode
ORDER BY fc.cntrycode;
