Original: select cntrycode, count(*) as numcust, sum(c_acctbal) as totacctbal from ( select substring(c_phone from 1 for 2) as cntrycode, c_acctbal from customer where substring(c_phone from 1 for 2) in ('24', '31', '26', '34', '30', '22', '14') and c_acctbal > ( select avg(c_acctbal) from customer where c_acctbal > 0.00 and substring(c_phone from 1 for 2) in ('24', '31', '26', '34', '30', '22', '14') ) and not exists ( select * from orders where o_custkey = c_custkey ) ) as custsale group by cntrycode order by cntrycode;
Rewritten: WITH eligible_countries AS (
  SELECT substring(c_phone from 1 for 2) AS cntrycode,
         c_custkey,
         c_acctbal
  FROM customer
  WHERE substring(c_phone from 1 for 2) IN ('24', '31', '26', '34', '30', '22', '14')
),
avg_balance AS (
  SELECT avg(c_acctbal) AS avg_acctbal
  FROM customer
  WHERE c_acctbal > 0.00
    AND substring(c_phone from 1 for 2) IN ('24', '31', '26', '34', '30', '22', '14')
),
no_orders_customers AS (
  SELECT DISTINCT o_custkey
  FROM orders
  WHERE o_custkey IS NOT NULL
)
SELECT ec.cntrycode,
       count(*) AS numcust,
       sum(ec.c_acctbal) AS totacctbal
FROM eligible_countries ec
CROSS JOIN avg_balance ab
LEFT JOIN no_orders_customers noc ON ec.c_custkey = noc.o_custkey
WHERE ec.c_acctbal > ab.avg_acctbal
  AND noc.o_custkey IS NULL
GROUP BY ec.cntrycode
ORDER BY ec.cntrycode;
