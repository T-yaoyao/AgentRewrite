Rewritten: SELECT p_brand, p_type, p_size, COUNT(DISTINCT ps_suppkey) AS supplier_cnt
FROM (
    SELECT ps_partkey, ps_suppkey
    FROM partsupp
    WHERE ps_suppkey NOT IN (
        SELECT s_suppkey
        FROM supplier
        WHERE s_comment LIKE '%Customer%Complaints%'
    )
) AS filtered_ps
JOIN (
    SELECT p_partkey, p_brand, p_type, p_size
    FROM part
    WHERE p_brand <> 'Brand#43'
      AND p_type NOT LIKE 'PROMO PLATED%'
      AND p_size IN (18, 8, 33, 17, 27, 6, 1, 50)
) AS filtered_part
ON filtered_ps.ps_partkey = filtered_part.p_partkey
GROUP BY p_brand, p_type, p_size
ORDER BY supplier_cnt DESC, p_brand, p_type, p_size;
