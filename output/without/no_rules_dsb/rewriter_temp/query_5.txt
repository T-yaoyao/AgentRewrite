customer_return_filtered AS (
  SELECT 
    sr_customer_sk AS ctr_customer_sk,
    sr_store_sk AS ctr_store_sk,
    sr_reason_sk AS ctr_reason_sk,
    SUM(sr_return_amt) AS ctr_total_return
  FROM store_returns
  INNER JOIN date_dim ON sr_returned_date_sk = d_date_sk
  WHERE d_year = 2001
    AND sr_return_amt / NULLIF(sr_return_quantity, 0) BETWEEN 182 AND 241
  GROUP BY sr_customer_sk, sr_store_sk, sr_reason_sk
)
SELECT c_customer_id
FROM customer_return_filtered crf
INNER JOIN store_avg sa ON crf.ctr_store_sk = sa.sr_store_sk
INNER JOIN store s ON s.s_store_sk = crf.ctr_store_sk
INNER JOIN customer c ON crf.ctr_customer_sk = c.c_customer_sk
INNER JOIN customer_demographics cd ON c.c_current_cdemo_sk = cd.cd_demo_sk
WHERE crf.ctr_total_return > sa.threshold
