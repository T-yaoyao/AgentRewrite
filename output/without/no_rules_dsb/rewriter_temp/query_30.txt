Rewritten: WITH filtered_dates AS (
  SELECT d_date_sk
  FROM date_dim
  WHERE d_year = 1999 AND d_moy = 7
),
filtered_items AS (
  SELECT i_item_sk, i_brand_id, i_brand, i_manufact_id, i_manufact
  FROM item
  WHERE i_category = 'Books'
),
filtered_customers AS (
  SELECT c.c_customer_sk, ca.ca_zip
  FROM customer c
  INNER JOIN customer_address ca ON c.c_current_addr_sk = ca.ca_address_sk
  WHERE c.c_birth_month = 10 AND ca.ca_state = 'IN'
),
filtered_stores AS (
  SELECT s_store_sk, s_zip
  FROM store
)
SELECT
  fi.i_brand_id AS brand_id,
  fi.i_brand AS brand,
  fi.i_manufact_id,
  fi.i_manufact,
  SUM(ss.ss_ext_sales_price) AS ext_price
FROM store_sales ss
INNER JOIN filtered_dates fd ON ss.ss_sold_date_sk = fd.d_date_sk
INNER JOIN filtered_items fi ON ss.ss_item_sk = fi.i_item_sk
INNER JOIN filtered_customers fc ON ss.ss_customer_sk = fc.c_customer_sk
INNER JOIN filtered_stores fs ON ss.ss_store_sk = fs.s_store_sk
WHERE ss.ss_wholesale_cost BETWEEN 57 AND 77
  AND substring(fc.ca_zip, 1, 5) <> substring(fs.s_zip, 1, 5)
GROUP BY fi.i_brand, fi.i_brand_id, fi.i_manufact_id, fi.i_manufact
ORDER BY ext_price DESC, fi.i_brand, fi.i_brand_id, fi.i_manufact_id, fi.i_manufact
LIMIT 100
