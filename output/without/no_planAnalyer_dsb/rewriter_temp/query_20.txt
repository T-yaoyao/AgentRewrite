  SELECT i_item_sk, i_brand_id, i_class_id, i_category_id

  LIMIT 1
),

  SELECT d_week_seq
  FROM date_dim
  WHERE d_year = 1998 AND d_moy = 12 AND d_dom = 13
  LIMIT 1
),
date_range AS (
  SELECT d_date_sk, d_week_seq
  FROM date_dim
  WHERE d_year BETWEEN 1998 AND 2000
),
filtered_item_web AS (
  SELECT i_item_sk, i_brand_id, i_class_id, i_category_id
  FROM item
  WHERE i_category IN ('Books', 'Home', 'Sports')
    AND i_manager_id BETWEEN 14 AND 23
),
cross_items AS (
  SELECT DISTINCT i_brand_id, i_class_id, i_category_id
  FROM (
    SELECT fi.i_brand_id, fi.i_class_id, fi.i_category_id, 'store' AS channel
    FROM store_sales ss
    JOIN filtered_item fi ON ss.ss_item_sk = fi.i_item_sk
    JOIN date_range dr ON ss.ss_sold_date_sk = dr.d_date_sk
    WHERE ss.ss_wholesale_cost BETWEEN 78 AND 98
    UNION ALL
    SELECT fi.i_brand_id, fi.i_class_id, fi.i_category_id, 'catalog' AS channel
    FROM catalog_sales cs
    JOIN filtered_item fi ON cs.cs_item_sk = fi.i_item_sk
    JOIN date_range dr ON cs.cs_sold_date_sk = dr.d_date_sk
    WHERE cs.cs_wholesale_cost BETWEEN 78 AND 98
    UNION ALL
    SELECT fi.i_brand_id, fi.i_class_id, fi.i_category_id, 'web' AS channel
    FROM web_sales ws
    JOIN filtered_item_web fi ON ws.ws_item_sk = fi.i_item_sk
    JOIN date_range dr ON ws.ws_sold_date_sk = dr.d_date_sk
    WHERE ws.ws_wholesale_cost BETWEEN 78 AND 98
  ) all_channels
  GROUP BY i_brand_id, i_class_id, i_category_id
  HAVING COUNT(DISTINCT channel) = 3
),
avg_sales_components AS (
  SELECT ss.ss_quantity * ss.ss_list_price AS sales_amount
  FROM store_sales ss
  JOIN date_range dr ON ss.ss_sold_date_sk = dr.d_date_sk
  WHERE ss.ss_wholesale_cost BETWEEN 78 AND 98
  UNION ALL
  SELECT cs.cs_quantity * cs.cs_list_price AS sales_amount
  FROM catalog_sales cs
  JOIN date_range dr ON cs.cs_sold_date_sk = dr.d_date_sk
  WHERE cs.cs_wholesale_cost BETWEEN 78 AND 98
  UNION ALL
  SELECT ws.ws_quantity * ws.ws_list_price AS sales_amount
  FROM web_sales ws
  JOIN date_range dr ON ws.ws_sold_date_sk = dr.d_date_sk
  WHERE ws.ws_wholesale_cost BETWEEN 78 AND 98
),
avg_sales AS (
  SELECT AVG(sales_amount) AS average_sales
  FROM avg_sales_components
),
this_year AS (
  SELECT 'store' AS channel,
         fi.i_brand_id,
         fi.i_class_id,
         fi.i_category_id,
         SUM(ss.ss_quantity * ss.ss_list_price) AS sales,
         COUNT(*) AS number_sales
  FROM store_sales ss
  JOIN filtered_item fi ON ss.ss_item_sk = fi.i_item_sk
  JOIN date_dim dd ON ss.ss_sold_date_sk = dd.d_date_sk
  JOIN cross_items ci ON fi.i_brand_id = ci.i_brand_id
                      AND fi.i_class_id = ci.i_class_id
                      AND fi.i_category_id = ci.i_category_id
  CROSS JOIN target_week_1999 tw
  WHERE dd.d_week_seq = tw.d_week_seq
    AND ss.ss_wholesale_cost BETWEEN 78 AND 98
  GROUP BY fi.i_brand_id, fi.i_class_id, fi.i_category_id
  HAVING SUM(ss.ss_quantity * ss.ss_list_price) > (SELECT average_sales FROM avg_sales)
),
last_year AS (
  SELECT 'store' AS channel,
         fi.i_brand_id,
         fi.i_class_id,
         fi.i_category_id,
         SUM(ss.ss_quantity * ss.ss_list_price) AS sales,
         COUNT(*) AS number_sales
  FROM store_sales ss
  JOIN filtered_item fi ON ss.ss_item_sk = fi.i_item_sk
  JOIN date_dim dd ON ss.ss_sold_date_sk = dd.d_date_sk
  JOIN cross_items ci ON fi.i_brand_id = ci.i_brand_id
                      AND fi.i_class_id = ci.i_class_id
                      AND fi.i_category_id = ci.i_category_id
  CROSS JOIN target_week_1998 tw
  WHERE dd.d_week_seq = tw.d_week_seq
    AND ss.ss_wholesale_cost BETWEEN 78 AND 98
  GROUP BY fi.i_brand_id, fi.i_class_id, fi.i_category_id
  HAVING SUM(ss.ss_quantity * ss.ss_list_price) > (SELECT average_sales FROM avg_sales)
)
SELECT this_year.channel AS ty_channel,
       this_year.i_brand_id AS ty_brand,
       this_year.i_class_id AS ty_class,
       this_year.i_category_id AS ty_category,
       this_year.sales AS ty_sales,
       this_year.number_sales AS ty_number_sales,
       last_year.channel AS ly_channel,
       last_year.i_brand_id AS ly_brand,
       last_year.i_class_id AS ly_class,
       last_year.i_category_id AS ly_category,
       last_year.sales AS ly_sales,
       last_year.number_sales AS ly_number_sales
FROM this_year
JOIN last_year ON this_year.i_brand_id = last_year.i_brand_id
               AND this_year.i_class_id = last_year.i_class_id
