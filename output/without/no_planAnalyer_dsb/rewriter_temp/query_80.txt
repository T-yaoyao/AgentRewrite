  CAST((revenue / 50) AS int) AS segment,
  COUNT(*) AS num_customers,
  CAST((revenue / 50) AS int) * 50 AS segment_base
FROM (
  SELECT
    c.c_customer_sk,
    SUM(ss.ss_ext_sales_price) AS revenue
  FROM
    (
      SELECT DISTINCT
        customer_sk,
        c_current_addr_sk
      FROM (
        SELECT
          cs.cs_bill_customer_sk AS customer_sk,
          c.c_current_addr_sk
        FROM catalog_sales cs
        JOIN date_dim d ON cs.cs_sold_date_sk = d.d_date_sk
          AND d.d_moy = 6
          AND d.d_year = 2001
        JOIN item i ON cs.cs_item_sk = i.i_item_sk
          AND i.i_category = 'Sports'
          AND i.i_class = 'camping'
        JOIN customer c ON cs.cs_bill_customer_sk = c.c_customer_sk
          AND c.c_birth_year BETWEEN 1932 AND 1945
        WHERE cs.cs_wholesale_cost BETWEEN 18 AND 48
        UNION ALL
        SELECT
          ws.ws_bill_customer_sk AS customer_sk,
          c.c_current_addr_sk
        FROM web_sales ws
        JOIN date_dim d ON ws.ws_sold_date_sk = d.d_date_sk
          AND d.d_moy = 6
          AND d.d_year = 2001
        JOIN item i ON ws.ws_item_sk = i.i_item_sk
          AND i.i_category = 'Sports'
          AND i.i_class = 'camping'
        JOIN customer c ON ws.ws_bill_customer_sk = c.c_customer_sk
          AND c.c_birth_year BETWEEN 1932 AND 1945
        WHERE ws.ws_wholesale_cost BETWEEN 18 AND 48
      ) union_result
    ) my_customers
  JOIN customer c ON my_customers.customer_sk = c.c_customer_sk
  JOIN customer_address ca ON c.c_current_addr_sk = ca.ca_address_sk
  JOIN store s ON ca.ca_county = s.s_county
    AND ca.ca_state = s.s_state
    AND s.s_state IN ('IA', 'MD', 'MO', 'MS', 'NE', 'NY', 'OK', 'TN', 'TX', 'VA')
  JOIN store_sales ss ON c.c_customer_sk = ss.ss_customer_sk
    AND ss.ss_wholesale_cost BETWEEN 18 AND 48
  JOIN date_dim d2 ON ss.ss_sold_date_sk = d2.d_date_sk
  CROSS JOIN date_range dr
  WHERE d2.d_month_seq BETWEEN dr.start_seq AND dr.end_seq
  GROUP BY c.c_customer_sk
) my_revenue
GROUP BY segment
ORDER BY segment, num_customers
LIMIT 100;
