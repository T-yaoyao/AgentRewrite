SELECT c_customer_sk, c_first_name, c_last_name, COUNT(*) AS cnt
FROM filtered_items fi
INNER JOIN store_sales ss ON fi.i_item_sk = ss.ss_item_sk
  AND ss.ss_sales_price / ss.ss_list_price BETWEEN 0.80 AND 1.00
  AND ss.ss_ticket_number IS NOT NULL
  AND ss.ss_customer_sk IS NOT NULL
INNER JOIN store_returns sr ON ss.ss_ticket_number = sr.sr_ticket_number

INNER JOIN filtered_customers fc ON ss.ss_customer_sk = fc.c_customer_sk
